// Icons const ICONS = { folder: ``, file: ``, video: ``, text: ``, pdf: ``, docx: ``, back: `` }; function getFileIcon(fileName) { const ext = fileName.split('.').pop().toLowerCase(); if (ext === 'mp4' || ext === 'webm' || ext === 'ogg') return ICONS.video; if (ext === 'pdf') return ICONS.pdf; if (ext === 'docx' || ext === 'doc') return ICONS.docx; if (ext === 'txt' || ext === 'md') return ICONS.text; return ICONS.file; }
